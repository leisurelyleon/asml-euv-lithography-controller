cmake_minimum_required(VERSION 3.24)
project(EUV_Optics_Simulation VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Optimization for the simulation engine
add_compile_options(-O3 -Wall -Wextra -pthread)

set(SIM_SOURCES
    src/WaferStageStepper.cpp
    src/ReticleMaskAlignment.cpp
    src/PhotonTracingEngine.cpp
)

add_executable(euv_optics_sim ${SIM_SOURCES})

target_include_directories(euv_optics_sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)

find_package(Threads REQUIRED)
target_link_libraries(euv_optics_sim PRIVATE Threads::Threads)
